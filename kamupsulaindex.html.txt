        // Kullanıcıdan gelen JS veya JSON Firebase Config metnini akıllı ayrıştırıcı
        function parseFirebaseConfigText(rawText) {
            let cleaned = rawText.trim();
            // "const firebaseConfig = " gibi önekleri temizle
            const firstBrace = cleaned.indexOf('{');
            const lastBrace = cleaned.lastIndexOf('}');
            if (firstBrace === -1 || lastBrace === -1) {
                throw new Error("Geçerli bir süslü parantez { ... } bloğu bulunamadı.");
            }
            cleaned = cleaned.substring(firstBrace, lastBrace + 1);

            // JS anahtarlarını çift tırnaklı geçerli JSON'a çevir (apiKey: -> "apiKey":)
            const jsonReady = cleaned
                .replace(/(['"])?([a-zA-Z0-9_]+)(['"])?\s*:/g, '"$2":')
                .replace(/'/g, '"')
                .replace(/,\s*}/g, '}');

            const configObj = JSON.parse(jsonReady);
            if (!configObj.apiKey || !configObj.projectId) {
                throw new Error("apiKey veya projectId alanları eksik.");
            }
            return configObj;
        }

        function connectAndSaveFirebase() {
            const rawInput = document.getElementById('firebaseConfigInput').value;
            if (!rawInput.trim()) {
                showToast("Eksik Bilgi", "Lütfen Firebase Console'dan kopyaladığınız config kodunu yapıştırın.");
                return;
            }

            try {
                const configObj = parseFirebaseConfigText(rawInput);
                localStorage.setItem('kp_v2_firebase_config', JSON.stringify(configObj));
                initFirebaseConnection(configObj, true);
            } catch (err) {
                showToast("Format Hatası", "Config kodu okunamadı: " + err.message);
            }
        }

        function initFirebaseConnection(configObj, isManualConnect) {
            try {
                if (unsubscribeSnapshot) {
                    unsubscribeSnapshot();
                    unsubscribeSnapshot = null;
                }

                // Mevcut Firebase uygulaması varsa silip yeniden başlat
                if (firebase.apps.length > 0) {
                    firebase.apps.forEach(app => app.delete());
                }

                firebase.initializeApp(configObj);
                db = firebase.firestore();

                // Gerçek zamanlı dinleyici (onSnapshot)
                unsubscribeSnapshot = db.collection('ilanlar').onSnapshot(
                    (snapshot) => {
                        isFirebaseLive = true;
                        updateDbStatusUI(true, configObj.projectId);

                        const cloudJobs = [];
                        snapshot.forEach(doc => {
                            cloudJobs.push({ id: doc.id, ...doc.data() });
                        });

                        allJobs = cloudJobs;
                        refreshPortalUI();

                        if (isManualConnect) {
                            showToast("Bulut Bağlantısı Başarılı!", "Firebase Firestore canlı olarak bağlandı.");
                            isManualConnect = false;
                        }
                    },
                    (error) => {
                        console.error("Firestore bağlantı hatası:", error);
                        isFirebaseLive = false;
                        updateDbStatusUI(false);
                        showToast("Veritabanı Yetki Uyarısı", "Firestore erişilemedi. Lütfen Test Modu kurallarını kontrol edin.");
                        loadLocalJobs();
                    }
                );
            } catch (err) {
                console.error(err);
                showToast("Bağlantı Hatası", "Firebase başlatılamadı.");
                loadLocalJobs();
            }
        }

        function disconnectFirebase() {
            if (unsubscribeSnapshot) {

        function getDaysRemaining(deadlineStr) {
            if (!deadlineStr) return 7;
            const today = new Date();
            today.setHours(0, 0, 0, 0);
            const target = new Date(deadlineStr);
            target.setHours(0, 0, 0, 0);
            const diffTime = target - today;
            const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
            return diffDays;
        }

        function formatTurkishDate(dateStr) {
            if (!dateStr) return "Belirtilmedi";
            const parts = dateStr.split('-');
            if (parts.length !== 3) return dateStr;
            const months = ["Ocak", "Şubat", "Mart", "Nisan", "Mayıs", "Haziran", "Temmuz", "Ağustos", "Eylül", "Ekim", "Kasım", "Aralık"];
            const day = parseInt(parts[2], 10);
            const monthName = months[parseInt(parts[1], 10) - 1] || parts[1];
            return `${day} ${monthName} ${parts[0]}`;
        }

        function isJobKpssExempt(job) {
            const k = (job.kpss || '').toLowerCase();
            return k.includes('şartsız') || k.includes('sartsiz') || k.includes('kura') || k.includes('yok');
        }

        function extractMinScore(kpssStr) {
            const match = (kpssStr || '').match(/(\d{2,3})\s*puan/i);
            return match ? parseInt(match[1], 10) : 0;
        }

        function renderJobs() {
            const q = document.getElementById('searchInput').value.toLowerCase().trim();
            const cityVal = document.getElementById('cityFilter').value;
            const eduVal = document.getElementById('eduFilter').value;
            const typeVal = document.getElementById('typeFilter').value;
            const noKpssOnly = document.getElementById('noKpssToggle').checked;
            const sortVal = document.getElementById('sortSelect').value;

            let filtered = allJobs.filter(job => {
                if (showFavoritesOnly && !favoriteIds.includes(job.id)) return false;
                if (activeCategory !== 'ALL' && job.category !== activeCategory) return false;

                if (quickStatMode === 'NO_KPSS' && !isJobKpssExempt(job)) return false;
                if (quickStatMode === 'CLOSING_SOON') {
                    const d = getDaysRemaining(job.deadline);
                    if (d < 0 || d > 3) return false;
                }

                if (noKpssOnly && !isJobKpssExempt(job)) return false;

                if (kpssFilterLimit) {
                    const eduMatch = job.education === 'Tümü' || job.education === kpssFilterLimit.edu;
                    if (!eduMatch) return false;
                    if (!isJobKpssExempt(job)) {
                        const minReq = extractMinScore(job.kpss);
                        if (minReq > kpssFilterLimit.score) return false;
                    }
                }

                if (q) {
                    const codesStr = Array.isArray(job.codes) ? job.codes.join(' ') : (job.codes || '');
                    const hay = `${job.institution} ${job.title} ${job.city} ${codesStr} ${job.kpss}`.toLowerCase();
                    if (!hay.includes(q)) return false;
                }

                if (cityVal !== 'ALL') {
                    if (job.city !== 'Türkiye Geneli' && job.city !== cityVal) return false;
                }

                if (eduVal !== 'ALL') {
                    if (job.education !== 'Tümü' && job.education !== eduVal) return false;
                }

                if (typeVal !== 'ALL' && job.recruitType !== typeVal) return false;

                return true;
            });

            // Sıralama
            if (sortVal === 'deadline') {
                filtered.sort((a, b) => getDaysRemaining(a.deadline) - getDaysRemaining(b.deadline));
            } else if (sortVal === 'quota') {
                filtered.sort((a, b) => (parseInt(b.quota, 10) || 0) - (parseInt(a.quota, 10) || 0));
            } else {
                filtered.reverse();
            }

            // Başlık ve Filtre Rozeti Güncelleme
            const heading = document.getElementById('listHeading');
            const badge = document.getElementById('activeFilterBadge');

            if (showFavoritesOnly) {
                heading.textContent = "Kaydettiğiniz Favori Kamu İlanları";
                badge.textContent = "Favorilerim";
                badge.classList.remove('hidden');
            } else if (kpssFilterLimit) {
                heading.textContent = `${kpssFilterLimit.score.toFixed(2)} KPSS Puanınıza Uygun İlanlar`;
                badge.textContent = "KPSS Robotu Aktif";
                badge.classList.remove('hidden');
            } else if (quickStatMode !== 'ALL') {
                const labels = { QUOTA: "En Yüksek Kontenjanlı Alımlar", NO_KPSS: "KPSS Şartsız Kamu İlanları", CLOSING_SOON: "Son 3 Günü Kalan İlanlar" };
                heading.textContent = labels[quickStatMode];
                badge.textContent = "Hızlı Filtre";
                badge.classList.remove('hidden');
            } else {
                heading.textContent = "Güncel Kamu Personeli Alım İlanları";
                badge.classList.add('hidden');
            }

            document.getElementById('visibleCount').textContent = filtered.length;

            const container = document.getElementById('jobsContainer');
            const emptyState = document.getElementById('emptyState');

            if (filtered.length === 0) {
                container.innerHTML = '';
                emptyState.classList.remove('hidden');
            } else {
                emptyState.classList.add('hidden');
                container.innerHTML = filtered.map(job => createJobCardHTML(job)).join('');
            }

            lucide.createIcons();
        }

        function calculateKpss() {
            const selectedEdu = document.querySelector('input[name="kpssEdu"]:checked')?.value || 'Lisans';
            latestCalcEdu = selectedEdu;

            let gyD = Math.min(60, Math.max(0, parseInt(document.getElementById('gyDogru').value, 10) || 0));
            let gyY = Math.min(60 - gyD, Math.max(0, parseInt(document.getElementById('gyYanlis').value, 10) || 0));
            let gkD = Math.min(60, Math.max(0, parseInt(document.getElementById('gkDogru').value, 10) || 0));
            let gkY = Math.min(60 - gkD, Math.max(0, parseInt(document.getElementById('gkYanlis').value, 10) || 0));

            const gyNet = Math.max(0, gyD - gyY * 0.25);
            const gkNet = Math.max(0, gkD - gkY * 0.25);
            const totalNet = gyNet + gkNet;

            const base = { Lisans: 46.2, Önlisans: 47.5, Ortaöğretim: 48.1 }[selectedEdu];
            const mult = { Lisans: 0.43, Önlisans: 0.425, Ortaöğretim: 0.42 }[selectedEdu];
            const score = totalNet === 0 ? 0 : Math.min(100, base + totalNet * mult);
            latestCalcScore = score;

            const typeMap = { Lisans: 'KPSS P3', Önlisans: 'KPSS P93', Ortaöğretim: 'KPSS P94' };
            document.getElementById('gyNetText').textContent = `${gyNet.toFixed(2)} Net`;
            document.getElementById('gkNetText').textContent = `${gkNet.toFixed(2)} Net`;
            document.getElementById('kpssResultLabel').textContent = `Tahmini ${typeMap[selectedEdu]} (${selectedEdu}) Puanınız`;
            document.getElementById('kpssScoreDisplay').textContent = score.toFixed(3);
            document.getElementById('kpssTotalNetDisplay').textContent = `(${totalNet.toFixed(2)} Net)`;

            const eligibleCount = allJobs.filter(j => {
                const eduOk = j.education === 'Tümü' || j.education === selectedEdu;
                if (!eduOk) return false;
                if (isJobKpssExempt(j)) return true;
                return extractMinScore(j.kpss) <= score;
            }).length;

            document.getElementById('kpssMatchHint').textContent = `Bu puan ile şu an ${eligibleCount} ilana başvurabilirsiniz!`;
        }

        function switchAdminTab(tabName) {
            const tabs = {
                form: { btn: 'tabBtnForm', panel: 'adminTabForm' },
                manage: { btn: 'tabBtnManage', panel: 'adminTabManage' },
                settings: { btn: 'tabBtnSettings', panel: 'adminTabSettings' }
            };

            Object.keys(tabs).forEach(key => {
                const isTarget = key === tabName;
                document.getElementById(tabs[key].panel).classList.toggle('hidden', !isTarget);
                const btn = document.getElementById(tabs[key].btn);
                if (isTarget) {
                    btn.className = "admin-tab px-4 py-2.5 rounded-t-xl text-xs sm:text-sm font-extrabold bg-white text-gov-800 border-t border-x border-slate-200 whitespace-nowrap flex items-center gap-1.5";
                } else {
                    btn.className = "admin-tab px-4 py-2.5 rounded-t-xl text-xs sm:text-sm font-bold text-slate-600 hover:text-navy-900 whitespace-nowrap flex items-center gap-1.5";
                }
            });

            if (tabName === 'manage') renderAdminManageList();
            lucide.createIcons();
        }

        async function handleSaveJob(e) {
            e.preventDefault();
            const editId = document.getElementById('editJobId').value;
            const codesRaw = document.getElementById('fCodes').value;
            const codesArray = codesRaw.split(',').map(s => s.trim()).filter(Boolean);

            const jobPayload = {
                institution: document.getElementById('fInstitution').value.trim(),
                title: document.getElementById('fTitle').value.trim(),
                category: document.getElementById('fCategory').value,
                recruitType: document.getElementById('fRecruitType').value,
                city: document.getElementById('fCity').value,
                education: document.getElementById('fEducation').value,
                kpss: document.getElementById('fKpss').value.trim(),
                quota: parseInt(document.getElementById('fQuota').value, 10) || 1,
                deadline: document.getElementById('fDeadline').value,
                codes: codesArray,
                applyUrl: document.getElementById('fUrl').value.trim(),
                details: document.getElementById('fDetails').value.trim()
            };

            if (isFirebaseLive && db) {
                try {
                    if (editId) {
                        await db.collection('ilanlar').doc(editId).set(jobPayload);
                        showToast("İlan Güncellendi", "Değişiklikler canlı bulut veritabanına kaydedildi.");
                    } else {
                        await db.collection('ilanlar').add(jobPayload);
                        showToast("İlan Yayınlandı", "Yeni ilan tüm ziyaretçiler için canlıya alındı.");
                    }
                    resetAdminForm();
                    switchAdminTab('manage');
                } catch (err) {
                    showToast("Bulut Kayıt Hatası", err.message);
                }
            } else {
                if (editId) {
                    const idx = allJobs.findIndex(j => j.id === editId);
                    if (idx !== -1) {
                        allJobs[idx] = { id: editId, ...jobPayload };
                    }
                    showToast("İlan Güncellendi", "İlan yerel depolamada güncellendi.");
                } else {
                    const newId = 'job_' + Date.now();
                    allJobs.push({ id: newId, ...jobPayload });
                    showToast("İlan Eklendi", "Yeni ilan başarıyla listeye eklendi.");
                }
                saveLocalJobs();
                resetAdminForm();
                switchAdminTab('manage');
            }
        }

        function resetAdminForm() {

            lucide.createIcons();
        }

        async function deleteJob(jobId) {
            if (isFirebaseLive && db) {
                try {
                    await db.collection('ilanlar').doc(jobId).delete();
                    showToast("İlan Silindi", "İlan bulut veritabanından kaldırıldı.");
                } catch (err) {
                    showToast("Silme Hatası", err.message);
                }
            } else {
                allJobs = allJobs.filter(j => j.id !== jobId);
                saveLocalJobs();
                showToast("İlan Silindi", "İlan listeden kaldırıldı.");
            }
        }

        async function seedDefaultJobsToDb() {
            if (isFirebaseLive && db) {
                try {
                    const batch = db.batch();
                    DEFAULT_JOBS.forEach(item => {
                        const { id, ...cleanData } = item;
                        const docRef = db.collection('ilanlar').doc(id);
                        batch.set(docRef, cleanData);
                    });
                    await batch.commit();
                    showToast("Örnek İlanlar Yüklendi", "10 adet örnek kamu ilanı Firebase bulut veritabanına aktarıldı.");
                } catch (err) {
                    showToast("Yükleme Hatası", err.message);
                }
            } else {
                allJobs = [...DEFAULT_JOBS];
                saveLocalJobs();
                showToast("Örnek İlanlar Yüklendi", "10 adet örnek kamu ilanı yerel listeye geri yüklendi.");
            }
        }

        function scrollToListings() {
            const el = document.getElementById('listingsSection');
            if (!el) return;

            window.scrollTo({ top: y, behavior: 'smooth' });
        }

        let toastTimer = null;
        function showToast(title, msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toastTitle').textContent = title;
            document.getElementById('toastMsg').textContent = msg;

            toast.classList.remove('translate-y-24', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            clearTimeout(toastTimer);
            toastTimer = setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-24', 'opacity-0');
            }, 3800);
        }
